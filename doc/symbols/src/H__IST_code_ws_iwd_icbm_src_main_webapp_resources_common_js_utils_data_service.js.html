 30 function DataMap(key_fiels, data_list) {

 32 	var _key_fiels = key_fiels;
 33 	var _data_map = {};
 34 	
 35 	var _setSource = function() {
 36 		_data_map = {};
 37 		for(var i=0; i<_data_list.length; i++) {
 38 			var key = 'dskey_'+ _data_list[i][_key_fiels];
 39 			_data_map[key] = _data_list[i];
 40 		}
 41 	}
 42 	
 43 	this.getList = function() {
 44 		return _data_list;
 45 	}
 46 	
 47 	this.getItem = function(key) {
 48 		return _data_map['dskey_'+ key];
 49 	}
 50 	
 51 	_setSource();
 52 }
