  3 	var options = $.extend({}, opt);
  4 	
  5 	options.async = (options.async===false?false:true);
  6 	options.type = (options.data?'POST':'GET');
  7 	options.data = (options.data?JSON.stringify(options.data):null);	// data to be sent to the server
  8 	options.contentType = 'application/json;charset=UTF-8';	// for send to server

 36 $.ajaxExcute = function(opt) {
 37 	var result;
 38 	var options = $.extend({}, opt);
 39 
 40 	var data = opt.data;
 41 	if( data && !(data instanceof Array) )
 42 		data = [data];
 43 	if(data)
 44 		data = JSON.stringify(data);
 45 	
 46 	options.async = (options.async?true:false);
 47 	options.type = (data?'POST':'GET');
 48 	options.data = data;
 49 	options.contentType = 'application/json;charset=UTF-8';
 50 	//contentType: "application/x-www-form-urlencoded; charset=UTF-8",
 51 	//options.url = options.url;
 52 	options.dataType = 'json';
 53 	options.success = function(response) {
 54 		result = {success: true, response: response};
 55 		if(opt.success)
 56 			opt.success(result);
 57 	};
 58 	options.error = function(err) {
 59 		result = {success: false, error: err};
 60 		if(opt.error)
 61 			opt.error(result);
 62 	};
 63 	
 64 	$.ajax(options);
 65 	return result;
 66 }

 94 ajaxUrl = function(sql_id, parmeters) {
 95 	var url = getContextPath() +'/query/json?qid=' + sql_id;
 96 	if(parmeters) {
 97 		for(var key in parmeters) url += '&'+ key +'='+ parmeters[key];
 98 	}
 99 	return url;

131 function _multilLoader(src_list, uuid) {
132 	var _sources = src_list;
133 	var _proc_index = 0;
134 
135 	var _internalSuccessHandler = function(transport, json) {
136 		var src = _sources[_proc_index++];
137 		var isLast = (_proc_index >= _sources.length)?true:false;
138 		var cbf = src.success;
139 		var sid = src.id;
140 		if(transport && cbf)
141 			cbf(sid, transport, isLast);
142 		
143 		if(isLast)
144 			dLoader.loaderContainer[uuid] = undefined;
145 		else
146 			_load();
147 	}
148 	var _internalErrorHandler = function(e) {
149 		var src = _sources[_proc_index++];
150 		var isLast = (_proc_index >= _sources.length)?true:false;
151 		
152 		src.error(src.id, e);
153 		dLoader.loaderContainer[uuid] = undefined;
154 	}
155 	
156 	var _load = function() {
157 		if(_proc_index >= _sources.length) return false;
158 		
159 		var src = _sources[_proc_index];
160 		$.ajaxSelect({
161 				url: src.url ? src.url : ajaxUrl(src.sql), 
162 				data:src.data, 
163 				success:_internalSuccessHandler, 
164 				error:src.error?_internalErrorHandler:null
165 			});
166 	}
167 
168 	_load();
169 }

235 _asyncLoader.prototype.loadPost = function() {
236 	if(this.interval) {
237 		this.state = 2;
238 		this.timer = setTimeout( this.loadAsync.bind(this), this.interval*1000);

241 		this.timer = 0;
242 		this.state = 0;

246 _asyncLoader.prototype.loadAsync = function() {
247 	
248 	if(this.timer) {
249 		clearTimeout(this.timer);
250 		this.timer = 0;
251 	}
252 	
253 	this.state = 1;
254 	$.ajaxSelect({
255 		url: this.url, 
256 		data: this.parameters, 
257 		success: this.successHandler, 
258 		error: ((this.error)?this.errorHandler:null)
259 	});
260 };
