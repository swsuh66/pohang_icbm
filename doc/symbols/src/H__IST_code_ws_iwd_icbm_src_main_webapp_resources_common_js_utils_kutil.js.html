  2 /**
  3  * @function dateFormat
  4  * @param {date} date
  5  * @param {string} mask format string
  6  * @param {boolean} utc
  7  * @description date-format class
  8  * (c) 2007-2009 Steven Levithan <stevenlevithan.com>
  9  * MIT license
 10  *
 11  * Includes enhancements by Scott Trenda <scott.trenda.net>
 12  * and Kris Kowal <cixar.com/~kris.kowal/>
 13  *
 14  * Accepts a date, a mask, or a date and a mask.
 15  * Returns a formatted version of the given date.
 16  * The date defaults to the current date/time.
 17  * The mask defaults to dateFormat.masks.default.
 18  */

 21 var dateFormat = function () {
 22 	var	token = /d{1,4}|m{1,4}|yy(?:yy)?|([HhMsTt])\1?|[LloSZ]|"[^"]*"|'[^']*'/g,
 23 		timezone = /\b(?:[PMCEA][SDP]T|(?:Pacific|Mountain|Central|Eastern|Atlantic) (?:Standard|Daylight|Prevailing) Time|(?:GMT|UTC)(?:[-+]\d{4})?)\b/g,
 24 		timezoneClip = /[^-+\dA-Z]/g,
 25 		pad = function (val, len) {
 26 			val = String(val);
 27 			len = len || 2;
 28 			while (val.length < len) val = "0" + val;
 29 			return val;
 30 		};
 31 
 32 	// Regexes and supporting functions are cached through closure
 33 	return function (date, mask, utc) {
 34 		var dF = dateFormat;
 35 
 36 		// You can't provide utc if you skip other args (use the "UTC:" mask prefix)
 37 		if (arguments.length == 1 && Object.prototype.toString.call(date) == "[object String]" && !/\d/.test(date)) {
 38 			mask = date;
 39 			date = undefined;
 40 		}
 41 
 42 		// Passing date through Date applies Date.parse, if necessary
 43 		date = date ? new Date(date) : new Date;
 44 		if (isNaN(date)) return '-';//throw SyntaxError("invalid date");
 45 
 46 		mask = String(dF.masks[mask] || mask || dF.masks["default"]);
 47 
 48 		// Allow setting the utc argument via the mask
 49 		if (mask.slice(0, 4) == "UTC:") {
 50 			mask = mask.slice(4);
 51 			utc = true;
 52 		}
 53 
 54 		var	_ = utc ? "getUTC" : "get",
 55 			d = date[_ + "Date"](),
 56 			D = date[_ + "Day"](),
 57 			m = date[_ + "Month"](),
 58 			y = date[_ + "FullYear"](),
 59 			H = date[_ + "Hours"](),
 60 			M = date[_ + "Minutes"](),
 61 			s = date[_ + "Seconds"](),
 62 			L = date[_ + "Milliseconds"](),
 63 			o = utc ? 0 : date.getTimezoneOffset(),
 64 			flags = {
 65 				d:    d,
 66 				dd:   pad(d),
 67 				ddd:  dF.i18n.dayNames[D],
 68 				dddd: dF.i18n.dayNames[D + 7],
 69 				m:    m + 1,
 70 				mm:   pad(m + 1),
 71 				mmm:  dF.i18n.monthNames[m],
 72 				mmmm: dF.i18n.monthNames[m + 12],
 73 				yy:   String(y).slice(2),
 74 				yyyy: y,
 75 				h:    H % 12 || 12,
 76 				hh:   pad(H % 12 || 12),
 77 				H:    H,
 78 				HH:   pad(H),
 79 				M:    M,
 80 				MM:   pad(M),
 81 				s:    s,
 82 				ss:   pad(s),
 83 				l:    pad(L, 3),
 84 				L:    pad(L > 99 ? Math.round(L / 10) : L),
 85 				t:    H < 12 ? "a"  : "p",
 86 				tt:   H < 12 ? "am" : "pm",
 87 				T:    H < 12 ? "A"  : "P",
 88 				//TT:   H < 12 ? "AM" : "PM",
 89 				TT:   H < 12 ? "오전" : "오후",
 90 				Z:    utc ? "UTC" : (String(date).match(timezone) || [""]).pop().replace(timezoneClip, ""),
 91 				o:    (o > 0 ? "-" : "+") + pad(Math.floor(Math.abs(o) / 60) * 100 + Math.abs(o) % 60, 4),
 92 				S:    ["th", "st", "nd", "rd"][d % 10 > 3 ? 0 : (d % 100 - d % 10 != 10) * d % 10]
 93 			};
 94 
 95 		return mask.replace(token, function ($0) {
 96 			return $0 in flags ? flags[$0] : $0.slice(1, $0.length - 1);
 97 		});
 98 	};
 99 }();

184 /**
185  * @method kutil.truncDate
186  * @description Date 를 년/월/일/시간 단위로 절사해서 반환. 
187  * @param {Date} date
188  * @param {String} val 절사 단위 ('y'/'m'/'d'/'h') *소문자*
189  * @returns {Date} 
190  */
191 kutil.truncDate = function (date, val) {
192     var d = new Date(date.valueOf());
193     switch(val) {
194     case 'y':
195         d.setHours(0, 0, 0, 0);
196         d.setDate(1);
197         d.setMonth(0);
198         break;
199     case 'm':
200         d.setHours(0, 0, 0, 0);
201         d.setDate(1);
202         break;
203     case 'd':
204         d.setHours(0, 0, 0, 0);
205         break;
206     case 'h':
207         d.setHours(date.getHours(), 0, 0, 0);
208         break;
209     }
210     return d;
211 }

245 kutil.isValidDate = function(dateString) {
246   var regEx = /^\d{4}-\d{1,2}-\d{1,2}$/;
247   if(!dateString.match(regEx)) return false;  // Invalid format
248   var d = new Date(dateString);
249   if(!d.getTime()) return false; // Invalid date (or this could be epoch)
250   return d.toISOString().slice(0,10) === dateString;
251 }

253 var _MS_PER_DAY = 1000 * 60 * 60 * 24;
254 
255 // a and b are javascript Date objects
256 kutil.dateDiffInDays = function(a, b) {
257   // Discard the time and time-zone information.
258   var utc1 = Date.UTC(a.getFullYear(), a.getMonth(), a.getDate());
259   var utc2 = Date.UTC(b.getFullYear(), b.getMonth(), b.getDate());
260 
261   return Math.floor( Math.abs(utc2 - utc1) / _MS_PER_DAY);
262 }

276 kutil.formatDateTime = function(tm) {
277 	var ret = kutil.formatDate(tm);
278 	if(!ret) return ret;
279 	if(tm.length > 8)
280 		ret += ' '+ tm.substr(8, 2);
281 	if(tm.length > 10)
282 		ret += ':'+ tm.substr(10, 2);
283 	if(tm.length > 12)
284 		ret += ':'+ tm.substr(12, 2);
285 	return ret;
286 }

297 kutil.countSub = function(list, fields, index) {
298 	var cnt = 1;
299 	var vals = [];
300 	for(var f=0; f<fields.length; f++){
301 		vals.push(list[index][fields[f]]);
302 	}
303 	for(var i=(index+1); i<list.length; i++) {
304 		for(var f=0; f<fields.length; f++){
305 			if(list[i][fields[f]] != vals[f]) return cnt;
306 		}
307 		cnt ++;
308 	}
309 	return cnt;
310 }

318 kutil.iframeWindow = function(iframe_object) {
319 	  var doc;
320 
321 	  if (iframe_object.contentWindow) {
322 	    return iframe_object.contentWindow;
323 	  }
324 
325 	  if (iframe_object.window) {
326 	    return iframe_object.window;
327 	  } 
328 
329 	  if (!doc && iframe_object.contentDocument) {
330 	    doc = iframe_object.contentDocument;
331 	  } 
332 
333 	  if (!doc && iframe_object.document) {
334 	    doc = iframe_object.document;
335 	  }
336 
337 	  if (doc && doc.defaultView) {
338 	   return doc.defaultView;
339 	  }
340 
341 	  if (doc && doc.parentWindow) {
342 	    return doc.parentWindow;
343 	  }
344 
345 	  return undefined;
346 	}
